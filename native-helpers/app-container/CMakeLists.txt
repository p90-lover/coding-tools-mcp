cmake_minimum_required(VERSION 3.20)
project(coding_tools_appcontainer LANGUAGES CXX)
add_executable(coding-tools-appcontainer main.cpp)
target_compile_features(coding-tools-appcontainer PRIVATE cxx_std_17)
target_compile_definitions(coding-tools-appcontainer PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(coding-tools-appcontainer PRIVATE userenv advapi32 shell32 pathcch)
if(MSVC)
  set_property(TARGET coding-tools-appcontainer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(coding-tools-appcontainer PRIVATE /W4 /WX /permissive- /EHsc)
  target_link_options(coding-tools-appcontainer PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
endif()
option(CODING_TOOLS_BUILD_PROBE "Build the synthetic boundary probe for release checks" OFF)
if(CODING_TOOLS_BUILD_PROBE)
  add_executable(coding-tools-sandbox-probe ../../aiTemp/completion/native_probe.cpp)
  target_compile_features(coding-tools-sandbox-probe PRIVATE cxx_std_17)
  target_link_libraries(coding-tools-sandbox-probe PRIVATE advapi32 ws2_32)
  if(MSVC)
    set_property(TARGET coding-tools-sandbox-probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()
